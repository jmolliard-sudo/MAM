-- Table principale CREATE TABLE IF NOT EXISTS attendance_records ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, date DATE NOT NULL UNIQUE, arrival_time TIME, departure_time TIME, notes TEXT, is_day_off BOOLEAN DEFAULT FALSE, updated_at TIMESTAMPTZ DEFAULT NOW() ); -- Mise à jour si la table existe déjà depuis une ancienne version ALTER TABLE attendance_records ADD COLUMN IF NOT EXISTS is_day_off BOOLEAN DEFAULT FALSE; -- Activer Realtime -- Realtime non nécessaire dans cette version REST/polling. -- La synchro entre appareils se fait par relecture automatique toutes les 5 secondes. -- Politique RLS ouverte (accès par URL) ALTER TABLE attendance_records ENABLE ROW LEVEL SECURITY; CREATE POLICY "acces_libre" ON attendance_records FOR ALL USING (true) WITH CHECK (true); -- Index performance CREATE INDEX IF NOT EXISTS idx_date ON attendance_records(date DESC);

-- Tables diversification alimentaire CREATE TABLE IF NOT EXISTS foods ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, category TEXT NOT NULL, name TEXT NOT NULL, min_age_months INTEGER, created_at TIMESTAMPTZ DEFAULT NOW(), UNIQUE(category, name) ); CREATE TABLE IF NOT EXISTS paul_foods ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, food_id UUID REFERENCES foods(id) ON DELETE CASCADE, already_eaten BOOLEAN DEFAULT FALSE, first_try_date DATE, notes TEXT, updated_at TIMESTAMPTZ DEFAULT NOW(), UNIQUE(food_id) ); ALTER TABLE foods ENABLE ROW LEVEL SECURITY; ALTER TABLE paul_foods ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "acces_libre_foods" ON foods; CREATE POLICY "acces_libre_foods" ON foods FOR ALL USING (true) WITH CHECK (true); DROP POLICY IF EXISTS "acces_libre_paul_foods" ON paul_foods; CREATE POLICY "acces_libre_paul_foods" ON paul_foods FOR ALL USING (true) WITH CHECK (true);
