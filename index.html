-- Table principale CREATE TABLE IF NOT EXISTS attendance_records ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, date DATE NOT NULL UNIQUE, arrival_time TIME, departure_time TIME, notes TEXT, updated_at TIMESTAMPTZ DEFAULT NOW() ); -- Activer Realtime -- Realtime non nécessaire dans cette version REST/polling. -- La synchro entre appareils se fait par relecture automatique toutes les 5 secondes. -- Politique RLS ouverte (accès par URL) ALTER TABLE attendance_records ENABLE ROW LEVEL SECURITY; CREATE POLICY "acces_libre" ON attendance_records FOR ALL USING (true) WITH CHECK (true); -- Index performance CREATE INDEX IF NOT EXISTS idx_date ON attendance_records(date DESC);
